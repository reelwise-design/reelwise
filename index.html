export default async function handler(req, res) { const token = process.env.TMDB_READ_ACCESS_TOKEN; if (!token) { return res.status(500).json({ error: "TMDB API token is not configured." }); } const { q, type, id, from, to } = req.query; async function tmdb(endpoint) { const response = await fetch( `https://api.themoviedb.org/3${endpoint}`, { headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" } } ); if (!response.ok) { const text = await response.text(); throw new Error( `TMDB request failed: ${response.status} ${text}` ); } return response.json(); } function clean(value) { return String(value || "") .toLowerCase() .replace(/[^a-z0-9]/g, ""); } /* * ----------------------------------------- * FIND AN ACTOR * ----------------------------------------- */ async function findActor(name) { const data = await tmdb( `/search/person?query=${encodeURIComponent( name )}&language=en-US&include_adult=false` ); const results = data.results || []; if (!results.length) { return null; } const exact = results.find( person => person.name && clean(person.name) === clean(name) ); return exact || results[0]; } /* * ----------------------------------------- * GET MOVIE CREDITS FOR AN ACTOR * ----------------------------------------- */ async function getMovieCredits(personId) { const data = await tmdb( `/person/${personId}/movie_credits?language=en-US` ); return (data.cast || []).filter(movie => { if (!movie.id || !movie.title) { return false; } const character = String( movie.character || "" ).toLowerCase(); /* * Ignore credits such as: * Self * Himself * Herself * Archive footage */ if ( character.includes("self") || character.includes("archive") ) { return false; } return true; }); } /* * ----------------------------------------- * GET THE CAST OF A MOVIE * ----------------------------------------- */ async function getMovieCast(movieId) { const data = await tmdb( `/movie/${movieId}/credits?language=en-US` ); return (data.cast || []).filter(person => { if (!person.id || !person.name) { return false; } const character = String( person.character || "" ).toLowerCase(); if ( character.includes("self") || character.includes("archive") ) { return false; } return true; }); } /* * ----------------------------------------- * BUILD ACTOR → MOVIES LOOKUP * ----------------------------------------- */ function buildMovieLookup(credits) { const lookup = new Map(); for (const movie of credits) { lookup.set(movie.id, movie); } return lookup; } /* * ----------------------------------------- * PRIORITIZE GOOD MOVIES * * This keeps the search from exploding * into thousands of obscure credits. * ----------------------------------------- */ function prioritizeMovies(credits) { return [...credits] .sort((a, b) => { const popularityA = Number(a.popularity || 0); const popularityB = Number(b.popularity || 0); const votesA = Number(a.vote_count || 0); const votesB = Number(b.vote_count || 0); const scoreA = popularityA + Math.min(votesA / 100, 50); const scoreB = popularityB + Math.min(votesB / 100, 50); return scoreB - scoreA; }) .slice(0, 80); } /* * ----------------------------------------- * SIX DEGREES BFS SEARCH * * A degree means one movie connection: * * Actor * ↓ * Movie * ↓ * Actor * * Maximum = 6 movies. * ----------------------------------------- */ async function findSixDegrees( startActor, targetActor ) { const MAX_DEGREES = 6; const startCredits = await getMovieCredits(startActor.id); const targetCredits = await getMovieCredits(targetActor.id); const startMovies = buildMovieLookup(startCredits); const targetMovies = buildMovieLookup(targetCredits); /* * FIRST: * Check whether they worked together directly. */ for (const movie of startCredits) { if (targetMovies.has(movie.id)) { return { distance: 1, path: [ { person: { id: startActor.id, name: startActor.name } }, { person: { id: targetActor.id, name: targetActor.name }, movie: { id: movie.id, title: movie.title, year: movie.release_date ? movie.release_date.substring(0, 4) : "" } } ] }; } } /* * ----------------------------------------- * BFS QUEUE * ----------------------------------------- */ const queue = []; queue.push({ actor: startActor, distance: 0, path: [ { person: { id: startActor.id, name: startActor.name } } ] }); const visitedActors = new Set([startActor.id]); /* * Cache movie casts so we don't repeatedly * ask TMDB for the same movie. */ const movieCastCache = new Map(); while (queue.length) { const current = queue.shift(); if ( current.distance >= MAX_DEGREES ) { continue; } /* * Get the current actor's movies. */ let credits; if ( current.actor.id === startActor.id ) { credits = startCredits; } else { credits = await getMovieCredits( current.actor.id ); } /* * Limit the number of movies examined at * each step so the request doesn't become * impossibly large. */ const movies = prioritizeMovies(credits); for (const movie of movies) { /* * Get cast. */ let cast; if (movieCastCache.has(movie.id)) { cast = movieCastCache.get(movie.id); } else { try { cast = await getMovieCast(movie.id); movieCastCache.set( movie.id, cast ); } catch { continue; } } /* * Look for target actor. */ const targetInMovie = cast.find( person => person.id === targetActor.id ); if (targetInMovie) { const newPath = [ ...current.path ]; /* * Add the actor we just reached. */ if ( current.actor.id !== startActor.id ) { newPath.push({ person: { id: current.actor.id, name: current.actor.name } }); } /* * Add target actor and movie. */ newPath.push({ person: { id: targetActor.id, name: targetActor.name }, movie: { id: movie.id, title: movie.title, year: movie.release_date ? movie.release_date.substring( 0, 4 ) : "" } }); return { distance: current.distance + 1, path: newPath }; } /* * ------------------------------------- * ADD NEW ACTORS TO THE BFS QUEUE * ------------------------------------- */ for (const person of cast) { if ( person.id === startActor.id || visitedActors.has(person.id) ) { continue; } visitedActors.add(person.id); const newPath = [ ...current.path ]; /* * Add the current actor if needed. */ if ( current.actor.id !== startActor.id ) { newPath.push({ person: { id: current.actor.id, name: current.actor.name } }); } /* * Add the connecting actor. */ newPath.push({ person: { id: person.id, name: person.name }, movie: { id: movie.id, title: movie.title, year: movie.release_date ? movie.release_date.substring( 0, 4 ) : "" } }); queue.push({ actor: { id: person.id, name: person.name }, distance: current.distance + 1, path: newPath }); } } } return null; } /* * ----------------------------------------- * SIX DEGREES ENDPOINT * ----------------------------------------- */ if (type === "degrees") { try { if (!from || !to) { return res.status(400).json({ error: "Enter two actors." }); } const [actorA, actorB] = await Promise.all([ findActor(from), findActor(to) ]); if (!actorA) { return res.status(404).json({ error: `Actor "${from}" was not found.` }); } if (!actorB) { return res.status(404).json({ error: `Actor "${to}" was not found.` }); } if ( actorA.id === actorB.id ) { return res.status(400).json({ error: "Choose two different actors." }); } const result = await findSixDegrees( actorA, actorB ); if (!result) { return res.status(404).json({ error: `No movie connection between ${actorA.name} and ${actorB.name} was found within six degrees.` }); } return res.status(200).json({ from: { id: actorA.id, name: actorA.name }, to: { id: actorB.id, name: actorB.name }, distance: result.distance, path: result.path }); } catch (error) { console.error( "Six Degrees error:", error ); return res.status(500).json({ error: error.message || "Six Degrees search failed." }); } } /* * ----------------------------------------- * NORMAL REELWISE SEARCH * ----------------------------------------- */ try { if (type === "movie") { return res.status(200).json( await tmdb( `/search/movie?query=${encodeURIComponent( q || "" )}&language=en-US` ) ); } if (type === "person") { return res.status(200).json( await tmdb( `/search/person?query=${encodeURIComponent( q || "" )}&language=en-US` ); } if (type === "movie-details") { return res.status(200).json( await tmdb( `/movie/${id}?language=en-US&append_to_response=credits` ) ); } if (type === "person-details") { return res.status(200).json( await tmdb( `/person/${id}?language=en-US&append_to_response=combined_credits` ) ); } return res.status(200).json( await tmdb( `/search/multi?query=${encodeURIComponent( q || "" )}&language=en-US` ) ); } catch (error) { console.error( "Reelwise search error:", error ); return res.status(500).json({ error: "Search failed." }); } }
