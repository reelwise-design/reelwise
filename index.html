function renderSearchResults(data){ const results = data.results || []; if(!results.length){ app.innerHTML = `

No results found

Try another movie or star.

`; return; } // Put stars first, then movies. // The API identifies them with result_type. const sortedResults = [...results].sort((a,b) => { if(a.result_type === "person" && b.result_type !== "person"){ return -1; } if(a.result_type !== "person" && b.result_type === "person"){ return 1; } return 0; }); app.innerHTML = `

Search Results

${sortedResults.map(item => { const isPerson = item.result_type === "person"; const title = isPerson ? item.name : item.title; const date = isPerson ? "Star" : (item.release_date || "").slice(0,4); return `
${escapeHtml(title)}
${escapeHtml(date)}
${isPerson ? "Star" : "Movie"}
`; }).join("")}
`; }
